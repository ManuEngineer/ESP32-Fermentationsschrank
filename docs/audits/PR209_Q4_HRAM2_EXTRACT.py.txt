#!/usr/bin/env python3
"""Sanitisierte Extraktion der H-RAM-2-Messdaten aus den privaten UART-Rohlogs (SSID/BSSID/IP bleiben privat).
Usage: extract.py EVIDENCE_DIR > out.tsv ; behaelt nur hram_* / app_main-resources-Zeilen (ohne Netzwerkkennungen)."""
import re, sys, os
base = sys.argv[1]
print("run\tt_s\tkind\tdetail")
for run in ["stack_B0","stack_B1","stack_B1u","stack_B2","stack_B3","stack_B4","stack_B5","stack_B5b","stack_B6","heap_run"]:
    for l in open(os.path.join(base, "runs", run, "uart_cap.log"), errors="replace"):
        m = re.match(r"\[\+\s*([\d.]+)s\] \w \(\d+\) (hram_http|app_main): (.*)", l.rstrip())
        if not m: continue
        d = m[3]
        if m[2] == "app_main" and not (d.startswith("resources: point=hram_30s") or d.startswith("hram_httpd_task")): continue
        d = re.sub(r"network_mode=\S+ network_state=\S+ ", "", d)
        print(f"{run}\t{m[1]}\t{m[2]}\t{d}")
