hexdec=[]
def parse_asm(path):
    frames={};cur=None;have=False
    for line in open(path,errors='replace'):
        m=re.match(r'^([0-9a-f]+) <([^>]+)>:',line)
        if m:
            cur=int(m[1],16);have=False;frames.setdefault(cur,None);continue
        if cur is not None and not have:
            h=re.match(r'^[0-9a-f]+:\t([0-9a-f]{8})\s*$',line)
            if h:
                w=int(h[1],16);b0,b1,b2=w&255,(w>>8)&255,(w>>16)&255
                if b0==0x36 and (b1&0xf)==1: frames[cur]=((b1>>4)|(b2<<4))*8;have=True;hexdec.append(cur)
                continue
            e=re.search(r'\bentry\s+a1,\s*(\d+)',line)
            if e: frames[cur]=int(e[1]);have=True
            elif re.search(r'\t(ret|ret\.n|retw|retw\.n)\b',line): frames[cur]=0;have=True
    return frames

def romframe(a):
    v=[r.get(a) for r in rom]
    if all(x is None for x in v): return None
    return max(x for x in v if x is not None)
