const { pw, open } = require('./common.js');
const fs = require('fs');
const R = []; const rec = (k, v) => { R.push([k, v]); console.log('##', k, '=>', typeof v === 'string' ? v.slice(0, 400) : JSON.stringify(v).slice(0, 400)); };
(async () => {
  const { b, p } = await open();
  const body = () => p.evaluate(() => document.body.innerText);
  await p.goto('http://<DEVICE_IP>/'); await p.waitForTimeout(1000);
  rec('login-shell', (await body()).replace(/\n+/g, ' | '));
  // wrong password (one attempt)
  await p.fill('input[type=password]', 'falsch-' + Date.now()); await p.click('button:has-text("Anmelden")'); await p.waitForTimeout(3500);
  await p.screenshot({ path: '03_wrongpw.png' });
  rec('wrong-password', (await body()).replace(/\n+/g, ' | '));
  await p.fill('input[type=password]', pw); await p.click('button:has-text("Anmelden")'); await p.waitForTimeout(4000);
  rec('after-login', (await body()).replace(/\n+/g, ' | '));
  for (const lang of ['de', 'en', 'es']) {
    await p.selectOption('#lang', lang); await p.waitForTimeout(800);
    const nav = await p.evaluate(() => [...document.querySelectorAll('button')].filter(e => e.offsetParent && !e.id).slice(0, 4).map(e => e.innerText));
    rec('nav-' + lang, nav);
    for (let i = 0; i < 4; i++) {
      await p.locator('button:not([id])').nth(i).click(); await p.waitForTimeout(900);
      await p.screenshot({ path: `v_${lang}_${i}.png` });
      rec(`view-${lang}-${i}-${nav[i]}`, (await body()).replace(/\n+/g, ' | '));
    }
  }
  await p.selectOption('#lang', 'de'); await p.waitForTimeout(500);
  rec('console', p.logs); await b.close();
})();
